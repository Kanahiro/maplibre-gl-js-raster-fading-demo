Reproduce issue: Reset to zoom 4, freeze time while zooming to 5 so the first tile opacity is 0, then advance 400 ms to check that the 300 ms fade finishes.

// Start with a fresh map at zoom 4 and raster-fade-duration: 300.
await map.once('idle');

const t = maplibregl.now();
maplibregl.setNow(t);

const firstDraw = new Promise(resolve => {
    function onRender() {
        if (!map.isSourceLoaded('osm')) return;
        map.off('render', onRender);
        resolve();
    }
    map.on('render', onRender);
});

map.setZoom(5);
await firstDraw;

maplibregl.setNow(t + 400);
for (let i = 0; i < 2; i++) {
    const rendered = map.once('render');
    map.triggerRepaint();
    await rendered;
}

maplibregl.restoreNow();
